cmake_minimum_required(VERSION 3.25)
project(bo3_zm_nx LANGUAGES C CXX)
get_filename_component(BO3_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(SPDLOG_NO_TZ_OFFSET)
set(REXGLUE_PLATFORM_SWITCH ON CACHE BOOL "Horizon runtime" FORCE)
set(REXGLUE_USE_VULKAN ON CACHE BOOL "Vulkan backend" FORCE)
set(REXGLUE_USE_D3D12 OFF CACHE BOOL "D3D12 backend" FORCE)
set(REXGLUE_ENABLE_TRACY OFF CACHE BOOL "Tracy" FORCE)
set(REXGLUE_ENABLE_PERF_COUNTERS OFF CACHE BOOL "Desktop counters" FORCE)
set(REXGLUE_SWITCH_NVK_SDK "${BO3_NX_MESA_SDK}" CACHE PATH "Horizon Mesa" FORCE)
add_subdirectory("${BO3_ROOT}/tools/nx-sdk" rexglue-sdk)
if(NOT EXISTS "${BO3_ROOT}/build/nx-generated/zombies/sources.cmake")
    message(FATAL_ERROR "Run scripts/nx/prepare-runtime.py first")
endif()
include("${BO3_ROOT}/build/nx-generated/zombies/sources.cmake")
set(BO3_GENERATED ${GENERATED_SOURCES})
# A huge straight-line registrar is only bootstrap code; GCC's O3 pass takes
# minutes and gigabytes here. Keep its ABI without optimizing this one file.
set_source_files_properties("${BO3_ROOT}/build/nx-generated/zombies/bo3_zm_register.cpp"
    PROPERTIES COMPILE_OPTIONS "-O0")
add_executable(bo3_zm_nx main.cpp ${BO3_GENERATED})
target_include_directories(bo3_zm_nx PRIVATE "${BO3_ROOT}/build/nx-generated/zombies" "${BO3_ROOT}/app/src")
target_compile_options(bo3_zm_nx PRIVATE -ffp-contract=off -fno-strict-aliasing -Wno-psabi)
target_link_libraries(bo3_zm_nx PRIVATE rex::runtime)
rexglue_configure_target(bo3_zm_nx GPU_PLUGINS xenos)
set_target_properties(bo3_zm_nx PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_link_options(bo3_zm_nx PRIVATE "-Wl,-Map,${CMAKE_BINARY_DIR}/bo3_zm_nx.map")
rexglue_switch_add_nro(bo3_zm_nx NAME "BO3 Zombies NX development" AUTHOR "Wiiucoop" VERSION "0.1.0")
